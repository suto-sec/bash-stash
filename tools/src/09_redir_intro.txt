@@ex 0926 redir_out_intro | Quick refresher: > (stdout to a file)
@@level 1
@@cmds >
@@readme
Save the output of `echo hola` into a file called `saludo.txt` (overwrite it). Nothing printed
on screen.
@@check
COMPARE="stdout exit files"
SEEDS=1
setup() { echo "old" > saludo.txt; }
@@solution
#!/bin/bash
echo hola > saludo.txt

@@ex 0928 redir_in_intro | Quick refresher: < (stdin from a file)
@@level 1
@@cmds <
@@readme
Print the number of lines of `datos.txt` using `wc -l`, feeding the file with **input
redirection** (`<`), not as an argument.
@@check
SEEDS=1
setup() { local i; for i in $(seq "$(randr 2 6)"); do word; done > datos.txt; }
extra_check() { ans_code | grep -qE 'wc[^<|]*datos\.txt' && fail "pass the file with < (not as an argument)"; }
@@solution
#!/bin/bash
wc -l < datos.txt
