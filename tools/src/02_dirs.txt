@@topic 02_directories | Directories & navigation
@@ex 0201 pwd_cd | pwd and cd
@@level 1
@@cmds cd, pwd
@@readme
The current directory contains the path `a/b/c`. Go into `a/b/c`, print the current directory,
then go back **up two levels** using `..` and print the current directory again.

```
/.../work/a/b/c
/.../work/a
```
@@check
SEEDS=1
setup() { mkdir -p a/b/c; }
extra_check() { must_use cd pwd; must_use '\.\.'; }
@@solution
#!/bin/bash
cd a/b/c
pwd
cd ../..
pwd

@@ex 0202 mkdir_tree | Build a directory tree

Create this structure **inside the current directory**:

```

├── Datos

│   └── Stocks
└── Textos
    ├── Cartas
```

The checker compares the resulting files and directories. Nothing needs to be printed.

@@ex 0209 ls_sort | ls sorting options
@@level 2
@@cmds ls -S, ls -t, ls -r
@@readme
The directory `data` contains files of different sizes and modification dates.
Print, one per line:

1. the file names ordered by **size**, biggest first
2. a line with `---`
3. the file names ordered by **modification time**, **oldest** first

Look at `ls -S`, `ls -t` and `ls -r`.
@@check
setup() {
  mkdir data; local i n
  for i in $(seq 6); do
    n="$(word)_$i.bin"; bigfile "data/$n" $(( $(randr 1 90) * 100 + i ))
    touch -d "2026-0$(randr 1 5)-$(randr 10 28) $(randr 10 23):00" "data/$n"
  done
}
@@solution
#!/bin/bash
ls -S data
echo ---
ls -tr data

@@ex 0213 du | Disk usage with du
@@level 3
@@cmds du -s, du -h, du -d
@@readme
The current directory contains `Datos` with several subdirectories and files.

Print:

1. the total size of `Datos` in **human readable** form (one line, `du` format)
2. the size of every directory **under** `Datos` with **max depth 1**, in KiB (`-k`)
3. the size of every directory **and file** under `Datos`, max depth 2, human readable

(Exactly what `du` prints; the checker runs your command and the reference on identical files.)
@@check
setup() { mkdir -p Datos/{a,b/c,d}; local f; for f in Datos/a/x Datos/b/y Datos/b/c/z Datos/d/w Datos/top; do bigfile "$f" $(( $(randr 1 40) * 1024 )); done; }
SORT_OUTPUT=1
extra_check() { must_use du; }
@@solution
#!/bin/bash
du -sh Datos
du -k -d 1 Datos
du -ah -d 2 Datos

@@ex 0214 absolute_vs_relative | Absolute and relative paths
@@level 3
@@cmds cd, pwd, realpath
@@readme
The file `where.txt` contains a **relative** path to an existing directory (e.g. `x/y/../z`).

Print:

1. the path exactly as written in the file
2. the **absolute, canonical** path of that directory (no `..`), obtained by going into it with
   `cd` and asking `pwd`
3. the path to go **from that directory back to the current one** as a relative path made only
   of `..` components separated by `/` (e.g. `../..`). The number of `..` equals the depth of the
   canonical relative path.

Hint: for 3 you may count the `/` in the canonical path relative to the start directory.
@@check
setup() {
  local a b c; a=$(word)1 b=$(word)2 c=$(word)3
  mkdir -p "$a/$b/$c" "$a/tmp"
  pick "$a/tmp/../$b/$c" "$a/$b/./$c" "$a/$b" "$a/tmp/../$b/$c/../$c" "$a" > where.txt
}
@@solution
#!/bin/bash
REL=$(cat where.txt)
echo "$REL"
START=$(pwd)
cd "$REL"
ABS=$(pwd)
echo "$ABS"
SUB=${ABS#"$START"/}
UP=..
for ((i = 0; i < $(echo "$SUB" | tr -cd / | wc -c); i++)); do UP="$UP/.."; done
echo "$UP"
