@@ex 0560 head_tail_intro | head and tail: first/last lines
@@level 1
@@cmds head -n, tail -n
@@readme
For the file `datos.txt`, print separated by a line `---`:

1. its first 3 lines
2. its last 2 lines
@@check
SEEDS=1
setup() { local i; for i in $(seq "$(randr 8 12)"); do echo "$i $(word)"; done > datos.txt; }
@@solution
#!/bin/bash
head -n 3 datos.txt
echo ---
tail -n 2 datos.txt

@@ex 0562 cut_intro | cut: extracting a field
@@level 1
@@cmds cut -d -f
@@readme
The file `datos.txt` has lines `name:age:city` (fields separated by `:`). Print only the second
field (the age) of every line.
@@check
SEEDS=1
setup() { local i; for i in $(seq "$(randr 5 10)"); do echo "$(word):$(randr 1 99):$(word)"; done > datos.txt; }
extra_check() { must_use cut; }
@@solution
#!/bin/bash
cut -d: -f2 datos.txt

@@ex 0564 uniq_intro | uniq: removing adjacent duplicates
@@level 1
@@cmds uniq
@@readme
`lista.txt` is already sorted and has some repeated adjacent lines. Print it with every
repetition collapsed into a single line (like `uniq` does).
@@check
SEEDS=1
setup() { local n w; n=$(randr 3 5); { for i in $(seq "$n"); do w=$(word); echo "$w"; echo "$w"; done; } | sort > lista.txt; }
extra_check() { must_use uniq; }
@@solution
#!/bin/bash
uniq lista.txt
