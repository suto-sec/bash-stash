@@step 1 | Common names
@@readme
Write `samenames.sh DIR1 DIR2`. It prints, one per line, the **names** of the regular files directly inside `DIR1` that also exist as regular files in `DIR2` (no subdirectories; the order does not matter). Names may contain spaces.
@@check
SORT_OUTPUT=1
ARGS=('left right' 'right left' 'left empty' 'left left')
COMPARE="stdout exit"
@@solution
#!/bin/bash
for f in "$1"/*; do
  [[ -f $f ]] || continue
  name=$(basename "$f")
  [[ -f $2/$name ]] && echo "$name"
done
exit 0

#!/bin/bash
n=0
for f in "$1"/*; do
  [[ -f $f ]] || continue
  name=$(basename "$f")
  if [[ -f $2/$name ]]; then echo "$name"; n=$((n + 1)); fi
done
echo "Common: $n"
@@step 3 | Check both directories
@@readme

@@solution
#!/bin/bash
if (( $# != 2 )); then echo "Error: two directories are needed" >&2; echo "Usage: $0 dir1 dir2" >&2; exit 1; fi
for d in "$1" "$2"; do
  [[ -e $d ]] || { echo "Error: $d does not exist" >&2; exit 2; }
  [[ -d $d ]] || { echo "Error: $d is not a directory" >&2; exit 3; }
done
n=0
for f in "$1"/*; do
  [[ -f $f ]] || continue
  name=$(basename "$f")
  if [[ -f $2/$name ]]; then echo "$name"; n=$((n + 1)); fi
done
echo "Common: $n"
