@@step 1 | Events per hour
@@readme
Every line of the log starts with the time `HH:MM:SS`. Write `timelog.sh LOG`. For each hour that has events print `HH: N` (the two-digit hour and how many lines), sorted by hour. An empty log prints nothing.

`${line:0:2}` is the hour; an associative array counts them.
@@check
ARGS=('events.log' 'one.log' 'empty.log')
COMPARE="stdout exit"
@@solution
#!/bin/bash
declare -A n
while IFS= read -r line; do
  h=${line:0:2}
  [[ -n $h ]] && n[$h]=$(( ${n[$h]:-0} + 1 ))
done < "$1"
for h in $(printf '%s\n' "${!n[@]}" | sort); do echo "$h: ${n[$h]}"; done
exit 0
@@step 2 | The busiest hour
@@readme
After the hours print `Busiest: HH` (the hour with the most events; with a tie the **earliest** hour). An empty log prints `Busiest: none`.
@@check
ARGS=('events.log' 'one.log' 'empty.log')
COMPARE="stdout exit"
@@solution
#!/bin/bash
declare -A n
while IFS= read -r line; do
  h=${line:0:2}
  [[ -n $h ]] && n[$h]=$(( ${n[$h]:-0} + 1 ))
done < "$1"
best= max=0
for h in $(printf '%s\n' "${!n[@]}" | sort); do
  echo "$h: ${n[$h]}"
  if (( ${n[$h]} > max )); then max=${n[$h]}; best=$h; fi
done
echo "Busiest: ${best:-none}"

  h=${line:0:2}

for h in $(printf '%s\n' "${!n[@]}" | sort); do
  echo "$h: ${n[$h]}"
  if (( ${n[$h]} > max )); then max=${n[$h]}; best=$h; fi
