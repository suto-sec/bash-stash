setup() {
  local i
  for ((i = 0; i < $(randr 5 12); i++)); do echo "$(randr 1 500)"; done > nums.txt
  printf '10\n9\n100\n-5\n7\n' > mixed.txt; echo 42 > one.txt; : > empty.txt
  echo x > locked.txt; chmod 000 locked.txt; mkdir adir

@@step 1 | Sort them
@@readme
`FILE` has one integer per line. Write `sortnums.sh FILE`: it prints the numbers sorted from the smallest to the largest (numerically: 9 comes before 10).

`sort -n` sorts numerically.
@@check
ARGS=('nums.txt' 'mixed.txt' 'one.txt')
COMPARE="stdout exit"
@@solution
#!/bin/bash
sort -n "$1"
@@step 2 | Smallest and largest
@@readme
After the sorted numbers print two lines, `min: X` and `max: Y`.

`head -n 1` and `tail -n 1` of the sorted list give the extremes.
@@check
ARGS=('nums.txt' 'mixed.txt' 'one.txt')
COMPARE="stdout exit"
@@solution
#!/bin/bash
sorted=$(sort -n "$1")
echo "$sorted"
echo "min: $(echo "$sorted" | head -n 1)"
echo "max: $(echo "$sorted" | tail -n 1)"
@@step 3 | Check the file
@@readme
Checks, in this order: not exactly one argument → error message **and the correct usage** (e.g. `Usage: sortnums.sh file`), exit **1**; not a readable regular file → error with its name, exit **2**; the file is empty → error message, exit **3**.
@@check
ARGS=('nums.txt' 'one.txt' '' 'nums.txt mixed.txt' 'nothing.txt' 'adir' 'locked.txt' 'empty.txt')
COMPARE="stdout exit errmsg"
extra_check() {
  [[ $REF_CODE == 1 ]] && { usage_ok || fail "the message should show the correct usage"; }
  [[ $REF_CODE == 2 ]] && mentions "$(eval "set -- $CASE"; echo "$1")"
}
@@solution
#!/bin/bash
if (( $# != 1 )); then echo "Error: one file is needed" >&2; echo "Usage: $0 file" >&2; exit 1; fi
[[ -f $1 && -r $1 ]] || { echo "Error: cannot read $1" >&2; exit 2; }
[[ -s $1 ]] || { echo "Error: $1 is empty" >&2; exit 3; }
sorted=$(sort -n "$1")
echo "$sorted"
echo "min: $(echo "$sorted" | head -n 1)"
echo "max: $(echo "$sorted" | tail -n 1)"
@@step 4 | Descending
@@readme
If the first argument is `-r`, print the list from the largest to the smallest (the `min:` and `max:` lines keep their meaning and stay at the end). `-r` alone is the usage error.
@@check
ARGS=('nums.txt' '-r nums.txt' '-r mixed.txt' '-r one.txt' '-r' '' 'nothing.txt' '-r empty.txt' '-r nothing.txt')
COMPARE="stdout exit errmsg"
extra_check() {
  [[ $REF_CODE == 1 ]] && { usage_ok || fail "the message should show the correct usage"; }
  [[ $REF_CODE == 2 ]] && mentions "$(eval "set -- $CASE"; echo "${@: -1}")"
}
@@solution
#!/bin/bash
order=-n
if [[ $1 == -r ]]; then order=-nr; shift; fi
if (( $# != 1 )); then echo "Error: one file is needed" >&2; echo "Usage: $0 [-r] file" >&2; exit 1; fi
[[ -f $1 && -r $1 ]] || { echo "Error: cannot read $1" >&2; exit 2; }
[[ -s $1 ]] || { echo "Error: $1 is empty" >&2; exit 3; }
sort $order "$1"
echo "min: $(sort -n "$1" | head -n 1)"
echo "max: $(sort -n "$1" | tail -n 1)"
