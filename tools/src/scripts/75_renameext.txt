@@step 1 | Rename by extension
@@readme
Write `renameext.sh DIR OLD NEW`. In `DIR` (not below it) every **regular file** whose name ends in `.OLD` is renamed so that it ends in `.NEW` (`OLD` and `NEW` are given **without** the dot). For each one print `oldname -> newname` (names only, no directory); the order does not matter. A directory called `dir.txt`, `x.txt.bak` or `notes.markdown` are **not** renamed (name must end exactly in `.OLD`). Hidden files count like any other. Names may contain spaces.

Example: `renameext.sh docs txt md` turns `b.txt` into `b.md` and prints `b.txt -> b.md`. In this step assume the new name is free.
@@check
SORT_OUTPUT=1
ARGS=('"my docs" txt md' 'empty txt md' 'docs bak old' 'docs markdown mkd' 'docs nothing x')
COMPARE="stdout exit files"
@@solution
#!/bin/bash
for f in "$1"/*."$2" "$1"/.*."$2"; do
  [[ -f $f ]] || continue
  name=$(basename "$f")
  new="${name%.$2}.$3"
  mv -- "$f" "$1/$new"
  echo "$name -> $new"
done
exit 0

#!/bin/bash
if (( $# != 3 )); then echo "Error: three arguments needed" >&2; echo "Usage: $0 dir old new" >&2; exit 1; fi
[[ -e $1 ]] || { echo "Error: $1 does not exist" >&2; exit 2; }
[[ -d $1 ]] || { echo "Error: $1 is not a directory" >&2; exit 3; }
n=0 skipped=0
for f in "$1"/*."$2" "$1"/.*."$2"; do
  [[ -f $f ]] || continue
  name=$(basename "$f")
  new="${name%.$2}.$3"
  if [[ -e $1/$new ]]; then echo "exists: $new" >&2; skipped=1; continue; fi
  mv -- "$f" "$1/$new"
  echo "$name -> $new"
  n=$((n + 1))
done
echo "Renamed $n files"
