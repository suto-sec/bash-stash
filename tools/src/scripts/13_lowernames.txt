@@script s13 | lowernames | Lower-case file names | lowernames.sh
@@level 3
@@tags files, copy and move, exit codes
@@cmds mv, tr / ${var,,}, test, for, counters
@@fixture
setup() {
  mkdir -p pics mixed empty
  echo 1 > pics/a.png; echo 2 > pics/b.png
  echo 1 > mixed/Photo.PNG; echo 2 > mixed/NOTES.txt; echo 3 > "mixed/My File.TXT"; echo 4 > mixed/ok.txt
  echo 5 > mixed/Report.doc; echo 6 > mixed/report.doc; mkdir mixed/SubDir
  echo x > notadir.txt
}
usage_ok() { [[ $ERR == *sage* || $ERR == *Uso* || $ERR == *uso* || $ERR == *lowernames.sh* ]]; }
@@step 1 | Rename to lower case
@@readme
Write `lowernames.sh DIR`. Every **regular file directly inside** `DIR` is renamed to its lower-case name (`NOTES.txt` becomes `notes.txt`); files already in lower case stay as they are. Subdirectories are not touched. Print `Renamed N files` (N = how many were renamed). Names may contain spaces.

`${name,,}` is `name` in lower case; `mv -- "$old" "$new"` renames. Assume no new name collides with an existing file in this step.
@@check
ARGS=('pics' 'empty' 'mixed/SubDir')
COMPARE="stdout exit files"
@@solution
#!/bin/bash
n=0
for f in "$1"/*; do
  [[ -f $f ]] || continue
  name=$(basename "$f"); low=${name,,}
  if [[ $name != "$low" ]]; then mv -- "$f" "$1/$low"; n=$((n + 1)); fi
done
echo "Renamed $n files"
@@step 2 | Mixed directory
@@readme
The directory may now contain upper-case names, spaces and subdirectories (`mixed` in the fixture has all three). The behaviour is the same, but test it on `mixed` too. If a lower-case name **already exists** (`Report.doc` and `report.doc`), do not overwrite it: print `skipped NAME` on standard error and do not count the file.
@@check
ARGS=('pics' 'mixed' 'empty')
COMPARE="stdout exit errmsg files"
@@solution
#!/bin/bash
n=0
for f in "$1"/*; do
  [[ -f $f ]] || continue
  name=$(basename "$f"); low=${name,,}
  [[ $name == "$low" ]] && continue
  if [[ -e $1/$low ]]; then echo "skipped $name" >&2; continue; fi
  mv -- "$f" "$1/$low"; n=$((n + 1))
done
echo "Renamed $n files"
@@step 3 | Check the directory
@@readme
Checks, in this order: not exactly one argument → error message **and the correct usage**, exit **1**; `DIR` does not exist → error with its name, exit **2**; it is not a directory → error with its name, exit **3**. Nothing is renamed or printed on standard output in those cases.
@@check
ARGS=('pics' 'mixed' '' 'pics mixed' 'nothing' 'notadir.txt')

@@solution
#!/bin/bash
if (( $# != 1 )); then echo "Error: one directory is needed" >&2; echo "Usage: $0 dir" >&2; exit 1; fi
[[ -e $1 ]] || { echo "Error: $1 does not exist" >&2; exit 2; }

n=0
for f in "$1"/*; do
  [[ -f $f ]] || continue
  name=$(basename "$f"); low=${name,,}
  [[ $name == "$low" ]] && continue
  if [[ -e $1/$low ]]; then echo "skipped $name" >&2; continue; fi
  mv -- "$f" "$1/$low"; n=$((n + 1))
done
echo "Renamed $n files"
@@step 4 | Exit code for skipped files
@@readme
If at least one file was skipped, finish (after printing the summary) with exit code **4**; otherwise 0.
@@check

@@solution
#!/bin/bash
if (( $# != 1 )); then echo "Error: one directory is needed" >&2; echo "Usage: $0 dir" >&2; exit 1; fi
[[ -e $1 ]] || { echo "Error: $1 does not exist" >&2; exit 2; }
[[ -d $1 ]] || { echo "Error: $1 is not a directory" >&2; exit 3; }
n=0 bad=0
for f in "$1"/*; do
  [[ -f $f ]] || continue
  name=$(basename "$f"); low=${name,,}
  [[ $name == "$low" ]] && continue
  if [[ -e $1/$low ]]; then echo "skipped $name" >&2; bad=1; continue; fi
  mv -- "$f" "$1/$low"; n=$((n + 1))
done
echo "Renamed $n files"
(( bad == 0 )) || exit 4
