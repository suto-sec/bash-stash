@@script s46 | initials | Initials of names | initials.sh
@@level 1
@@tags text, arguments, loops
@@cmds for, ${var:0:1}, ${var^^}, exit codes
@@fixture
setup() { :; }
usage_ok() { [[ $ERR == *sage* || $ERR == *Uso* || $ERR == *uso* || $ERR == *initials.sh* ]]; }
@@step 1 | One name
@@readme
Write `initials.sh NAME`. `NAME` is one argument with several words (`"ana maria ruiz"`). Print the **initials** of its words in upper case, with no separator: `AMR`.

`for w in $1; do ... ${w:0:1}` goes through the words (the unquoted `$1` is split at the blanks) and takes the first character.
@@check
ARGS=('"ana maria ruiz"' 'luis' '"Eva Gil"' '"juan  carlos   perez"' '""')
COMPARE="stdout exit"
@@solution
#!/bin/bash
out=
for w in $1; do
  out+=${w:0:1}
done
echo "${out^^}"
@@step 2 | Exactly one argument
@@readme
Not exactly one argument → error message **and the correct usage** (e.g. `Usage: initials.sh "full name"`) on standard error, exit **1**.
@@check
ARGS=('"ana maria ruiz"' 'luis' '' 'ana maria' '"a b" "c d"')
COMPARE="stdout exit errmsg"
extra_check() { [[ $REF_CODE == 1 ]] && { usage_ok || fail "the message should show the correct usage"; }; }
@@solution
#!/bin/bash
if (( $# != 1 )); then echo "Error: one name is needed" >&2; echo "Usage: $0 \"full name\"" >&2; exit 1; fi
out=
for w in $1; do
  out+=${w:0:1}
done
echo "${out^^}"
@@step 3 | Many names
@@readme
Accept **one or more** names (none is still the usage error). Print one line `NAME: INITIALS` for each, in order.
@@check
ARGS=('"ana maria ruiz"' '"ana maria" luis "Eva Gil"' 'x "a b c d"' '')
COMPARE="stdout exit errmsg"
extra_check() { [[ $REF_CODE == 1 ]] && { usage_ok || fail "the message should show the correct usage"; }; }
@@solution
#!/bin/bash
if (( $# == 0 )); then echo "Error: at least one name is needed" >&2; echo "Usage: $0 \"full name\"..." >&2; exit 1; fi
for name in "$@"; do
  out=
  for w in $name; do out+=${w:0:1}; done
  echo "$name: ${out^^}"
done
