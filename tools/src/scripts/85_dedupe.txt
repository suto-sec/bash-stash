@@step 1 | The extra copies
@@readme
Write `dedupe.sh DIR`. Two regular files are **duplicates** when their **content is identical** (the name does not matter; two empty files are duplicates). Look at every regular file below `DIR`, sort their paths (as `sort` sorts text); in each group of duplicates the **first path** is the original and every other path is a **copy**. Print the paths of the copies, **sorted**.

Hint: `md5sum` gives a hash per file; the files that share a hash are duplicates.
@@check
ARGS=('pics' '"my pics"' 'single' 'empty')
COMPARE="stdout exit"
@@solution
#!/bin/bash
find "$1" -type f -print0 | sort -z | xargs -0 md5sum | awk '{h=$1; sub(/^[^ ]+  /, ""); if (seen[h]++) print}' | sort
exit 0

#!/bin/bash

if [[ $1 == -d ]]; then del=1; shift; fi
if (( $# > 1 )); then echo "Error: too many arguments" >&2; echo "Usage: $0 [-d] [dir]" >&2; exit 1; fi

find "$dir" -type f -print0 | sort -z | xargs -0 -r md5sum | awk '{h=$1; sub(/^[^ ]+  /, ""); if (seen[h]++) print}' | sort | while IFS= read -r f; do

#!/bin/bash
del=
if [[ $1 == -d ]]; then del=1; shift; fi
if (( $# > 1 )); then echo "Error: too many arguments" >&2; echo "Usage: $0 [-d] [dir]" >&2; exit 1; fi
dir=${1:-.}
[[ -e $dir ]] || { echo "Error: $dir does not exist" >&2; exit 2; }
[[ -d $dir ]] || { echo "Error: $dir is not a directory" >&2; exit 3; }
n=0 bytes=0
while IFS= read -r f; do
  n=$((n + 1)); bytes=$((bytes + $(stat -c %s -- "$f")))
  if [[ -n $del ]]; then rm -- "$f"; echo "removed $f"; else echo "$f"; fi
done < <(find "$dir" -type f -print0 | sort -z | xargs -0 -r md5sum | awk '{h=$1; sub(/^[^ ]+  /, ""); if (seen[h]++) print}' | sort)
if [[ -n $del ]]; then echo "Removed $n files, freed $bytes bytes"; exit 0; fi
echo "$n duplicates, $bytes bytes"
(( n == 0 )) || exit 4
