@@script s32 | echoargs | Numbering the arguments | echoargs.sh
@@level 1
@@tags arguments, loops
@@cmds for, counters, shift, exit codes
@@fixture
setup() { :; }
usage_ok() { [[ $ERR == *sage* || $ERR == *Uso* || $ERR == *uso* || $ERR == *echoargs.sh* ]]; }
@@step 1 | Number them
@@readme
Write `echoargs.sh`. It prints each of its arguments on its own line, preceded by its position and a colon:

```
echoargs.sh red "dark blue"   ->   1: red
                                   2: dark blue
```

Keep a counter in the loop: `i=$((i + 1))`.
@@check
ARGS=('red' 'red green blue' '"dark blue" x' '')
COMPARE="stdout exit"
@@solution
#!/bin/bash
i=0
for a in "$@"; do
  i=$((i + 1))
  echo "$i: $a"
done
@@step 2 | At least one argument
@@readme
With **no arguments** print an error message **and the correct usage** (e.g. `Usage: echoargs.sh arg...`) on standard error and exit with code **1** (nothing on standard output).
@@check
ARGS=('red' 'red green blue' '"dark blue" x' '')
COMPARE="stdout exit errmsg"
extra_check() { [[ $REF_CODE == 1 ]] && { usage_ok || fail "the message should show the correct usage"; }; }
@@solution
#!/bin/bash
if (( $# == 0 )); then echo "Error: no arguments given" >&2; echo "Usage: $0 arg..." >&2; exit 1; fi
i=0
for a in "$@"; do
  i=$((i + 1))
  echo "$i: $a"
done
@@step 3 | Backwards
@@readme
If the **first** argument is `-r`, print the others **in reverse order** but keeping the positions they have in the original order: `echoargs.sh -r red green blue` prints `3: blue`, `2: green`, `1: red`. `-r` with nothing after it is the usage error (exit 1).
@@check
ARGS=('red' 'red green blue' '-r red green blue' '-r "dark blue" x' '-r one' '-r' '')
COMPARE="stdout exit errmsg"
extra_check() { [[ $REF_CODE == 1 ]] && { usage_ok || fail "the message should show the correct usage"; }; }
@@solution
#!/bin/bash
rev=
if [[ $1 == -r ]]; then rev=1; shift; fi
if (( $# == 0 )); then echo "Error: no arguments given" >&2; echo "Usage: $0 [-r] arg..." >&2; exit 1; fi
if [[ -n $rev ]]; then
  for ((i = $#; i >= 1; i--)); do echo "$i: ${!i}"; done
else
  i=0
  for a in "$@"; do i=$((i + 1)); echo "$i: $a"; done
fi
