@@script s43 | maxnum | The biggest number | maxnum.sh
@@level 1
@@tags tests, arithmetic, arguments
@@cmds for, if, arithmetic, [[ =~ ]], exit codes
@@fixture
setup() { :; }
usage_ok() { [[ $ERR == *sage* || $ERR == *Uso* || $ERR == *uso* || $ERR == *maxnum.sh* ]]; }
@@step 1 | Find the biggest
@@readme
Write `maxnum.sh N...`. It prints the biggest of the integers it receives (at least one, possibly negative): `maxnum.sh 3 9 4` prints `9`.

Start with the first number as the biggest so far, then compare each other one: `if (( n > max )); then max=$n; fi`.
@@check
ARGS=('5' '3 9 4' '-5 -2 -9' '10 10 10' '0 -1 7 7')
COMPARE="stdout exit"
@@solution
#!/bin/bash
max=$1
for n in "$@"; do
  if (( n > max )); then max=$n; fi
done
echo "$max"
@@step 2 | Check the numbers
@@readme
No arguments → error message **and the correct usage** (e.g. `Usage: maxnum.sh N...`) on standard error, exit **1**. Any argument that is not an integer (an optional `-` and digits) → error that includes it, exit **2**; check all of them before printing.
@@check
ARGS=('5' '3 9 4' '-5 -2' '' 'a' '3 x 4' '2.5' '--3')
COMPARE="stdout exit errmsg"
extra_check() {
  [[ $REF_CODE == 1 ]] && { usage_ok || fail "the message should show the correct usage"; }
  [[ $REF_CODE == 2 ]] && { local a; for a in $(eval "set -- $CASE"; echo "$@"); do [[ $a =~ ^-?[0-9]+$ ]] || { mentions "$a"; break; }; done; }
}
@@solution
#!/bin/bash
if (( $# == 0 )); then echo "Error: no numbers given" >&2; echo "Usage: $0 N..." >&2; exit 1; fi
for n in "$@"; do
  [[ $n =~ ^-?[0-9]+$ ]] || { echo "Error: '$n' is not an integer" >&2; exit 2; }
done
max=$1
for n in "$@"; do
  if (( n > max )); then max=$n; fi
done
echo "$max"
@@step 3 | And the smallest
@@readme
Print two lines instead of the single number, in this order: `max: X` (the biggest) and `min: Y` (the smallest). With a single number both are that number: `maxnum.sh 5` prints `max: 5` and `min: 5`. The checks of step 2 stay.
@@check
ARGS=('5' '3 9 4' '-5 -2 -9' '10 10 10' '' 'x 1')
COMPARE="stdout exit errmsg"
extra_check() {
  [[ $REF_CODE == 1 ]] && { usage_ok || fail "the message should show the correct usage"; }
  [[ $REF_CODE == 2 ]] && mentions "x"
}
@@solution
#!/bin/bash
if (( $# == 0 )); then echo "Error: no numbers given" >&2; echo "Usage: $0 N..." >&2; exit 1; fi
for n in "$@"; do
  [[ $n =~ ^-?[0-9]+$ ]] || { echo "Error: '$n' is not an integer" >&2; exit 2; }
done
max=$1 min=$1
for n in "$@"; do
  if (( n > max )); then max=$n; fi
  if (( n < min )); then min=$n; fi
done
echo "max: $max"
echo "min: $min"
