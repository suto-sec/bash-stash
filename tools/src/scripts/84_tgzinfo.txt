usage_ok() { [[ $ERR == *sage* || $ERR == *Uso* || $ERR == *uso* || $ERR == *tgzinfo.sh* ]]; }
@@step 1 | Count the entries
@@readme
Write `tgzinfo.sh FILE.tgz`. It prints three lines about the members of the gzip-compressed tar archive (`tar -tzf` lists them; a **directory** entry ends in `/`, including the archive's own `./`):

```
Entries: 12
Directories: 4
Files: 8
```
`Entries` counts every member; `Directories` those ending in `/`; `Files` the rest.
@@check
ARGS=('data.tgz' 'small.tgz' 'dirs.tgz')
COMPARE="stdout exit"
@@solution
#!/bin/bash
list=$(tar -tzf "$1")
total=$(echo "$list" | grep -c .)
dirs=$(echo "$list" | grep -c '/$')
echo "Entries: $total"
echo "Directories: $dirs"
echo "Files: $((total - dirs))"
@@step 2 | The usual checks
@@readme

@@step 3 | The largest file
@@readme
Add a fourth line `Largest: NAME (BYTES bytes)` with the **regular file** of the archive that has the most bytes (`tar -tzvf` shows sizes; `NAME` is the member name exactly as `tar -tzf` prints it, `./img/photo.jpg`). An archive with no regular file prints `Largest: none`. In these archives the largest file is unique and its name has no spaces.
@@check

@@solution
#!/bin/bash
if (( $# != 1 )); then echo "Error: one argument needed" >&2; echo "Usage: $0 file.tgz" >&2; exit 1; fi
[[ -e $1 ]] || { echo "Error: $1 does not exist" >&2; exit 2; }
[[ -f $1 ]] || { echo "Error: $1 is not a regular file" >&2; exit 3; }
list=$(tar -tzf "$1" 2>/dev/null) || { echo "Error: $1 is not a gzip tar archive" >&2; exit 4; }
total=$(echo "$list" | grep -c .)
dirs=$(echo "$list" | grep -c '/$')
echo "Entries: $total"
echo "Directories: $dirs"
echo "Files: $((total - dirs))"
big=$(tar -tzvf "$1" | awk '$1 ~ /^-/ {print $3, $6}' | sort -k1,1nr | head -n 1)
if [[ -z $big ]]; then echo "Largest: none"; else echo "Largest: ${big#* } (${big%% *} bytes)"; fi
@@step 4 | Extensions (-e)
@@readme
With `-e` as the **first** argument add, after those four lines, one line `.EXT: N` for every extension of the **regular files** (the text after the last dot of the member's **last path component**, with the dot; names without a dot count as `(none)`; `Makefile` is `(none)`), the most common first and equal counts in alphabetical order. The usage and all the checks stay (with `-e` the number of other arguments is still one).
@@check

extra_check() {

@@solution
#!/bin/bash
ext=
if [[ $1 == -e ]]; then ext=1; shift; fi
if (( $# != 1 )); then echo "Error: one argument needed" >&2; echo "Usage: $0 [-e] file.tgz" >&2; exit 1; fi
[[ -e $1 ]] || { echo "Error: $1 does not exist" >&2; exit 2; }
[[ -f $1 ]] || { echo "Error: $1 is not a regular file" >&2; exit 3; }
list=$(tar -tzf "$1" 2>/dev/null) || { echo "Error: $1 is not a gzip tar archive" >&2; exit 4; }
total=$(echo "$list" | grep -c .)
dirs=$(echo "$list" | grep -c '/$')
echo "Entries: $total"
echo "Directories: $dirs"
echo "Files: $((total - dirs))"
big=$(tar -tzvf "$1" | awk '$1 ~ /^-/ {print $3, $6}' | sort -k1,1nr | head -n 1)
if [[ -z $big ]]; then echo "Largest: none"; else echo "Largest: ${big#* } (${big%% *} bytes)"; fi
if [[ -n $ext ]]; then
  echo "$list" | grep -v '/$' | while IFS= read -r m; do
    b=${m##*/}
    if [[ $b == *.* ]]; then echo ".${b##*.}"; else echo "(none)"; fi
  done | sort | uniq -c | sort -k1,1nr -k2,2 | while read -r n e; do echo "$e: $n"; done
fi
exit 0
