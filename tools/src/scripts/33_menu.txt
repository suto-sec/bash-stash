@@script s33 | menu | A tiny command menu | menu.sh
@@level 2
@@tags case, arguments, text
@@cmds case, ${#var}, tr, shift, exit codes
@@fixture
setup() { :; }
usage_ok() { [[ $ERR == *sage* || $ERR == *Uso* || $ERR == *uso* || $ERR == *menu.sh* ]]; }
@@step 1 | One command
@@readme
Write `menu.sh COMMAND TEXT`. For the command `upper` it prints `TEXT` in upper case: `menu.sh upper hello` prints `HELLO`. (`tr a-z A-Z` or `${2^^}`.) Use `case "$1" in`.
@@check
ARGS=('upper hello' 'upper "two words"' 'upper ABC123')
COMPARE="stdout exit"
@@solution
#!/bin/bash
case $1 in
  upper) echo "${2^^}" ;;
esac
@@step 2 | More commands
@@readme
Add `lower` (lower case) and `len` (the number of characters of the text: `menu.sh len hello` prints `5`).
@@check
ARGS=('upper hello' 'lower "Two WORDS"' 'len hello' 'len ""' 'len "two words"' 'upper ABC123')
COMPARE="stdout exit"
@@solution
#!/bin/bash
case $1 in
  upper) echo "${2^^}" ;;
  lower) echo "${2,,}" ;;
  len) echo "${#2}" ;;
esac
@@step 3 | Unknown commands
@@readme
A command that is none of the three prints an error message that includes it on standard error and exits with code **2**. With no argument at all: error message **and the correct usage**, exit **1**.
@@check
ARGS=('upper hello' 'lower HELLO' 'len abc' 'reverse abc' 'UPPER abc' '' 'x')
COMPARE="stdout exit errmsg"
extra_check() {
  [[ $REF_CODE == 1 ]] && { usage_ok || fail "the message should show the correct usage"; }
  [[ $REF_CODE == 2 ]] && mentions "$(eval "set -- $CASE"; echo "$1")"
}
@@solution
#!/bin/bash
if (( $# == 0 )); then echo "Error: a command is needed" >&2; echo "Usage: $0 upper|lower|len text" >&2; exit 1; fi
case $1 in
  upper) echo "${2^^}" ;;
  lower) echo "${2,,}" ;;
  len) echo "${#2}" ;;
  *) echo "Error: unknown command '$1'" >&2; exit 2 ;;
esac
@@step 4 | The text is required
@@readme
A valid command needs **exactly one text argument** after it (an empty string `""` is a text). Otherwise: error message **and the correct usage**, exit code **3**. The unknown-command check comes first.
@@check
ARGS=('upper hello' 'len ""' 'lower "A B"' 'upper' 'len' 'upper a b' 'reverse abc' 'reverse' '')
COMPARE="stdout exit errmsg"
extra_check() {
  [[ $REF_CODE == 1 ]] && { usage_ok || fail "the message should show the correct usage"; }
  [[ $REF_CODE == 2 ]] && mentions "$(eval "set -- $CASE"; echo "$1")"
  [[ $REF_CODE == 3 ]] && { usage_ok || fail "the message should show the correct usage"; }
}
@@solution
#!/bin/bash
if (( $# == 0 )); then echo "Error: a command is needed" >&2; echo "Usage: $0 upper|lower|len text" >&2; exit 1; fi
case $1 in
  upper|lower|len) ;;
  *) echo "Error: unknown command '$1'" >&2; exit 2 ;;
esac
if (( $# != 2 )); then echo "Error: exactly one text is needed" >&2; echo "Usage: $0 upper|lower|len text" >&2; exit 3; fi
case $1 in
  upper) echo "${2^^}" ;;
  lower) echo "${2,,}" ;;
  len) echo "${#2}" ;;
esac
