@@step 1 | Files of a user
@@readme
Write `ownedby.sh USER DIR`. It prints the path (as `find` shows it) of every **regular file** below `DIR` that belongs to the user `USER` (`find DIR -type f -user USER`). The order does not matter. In the checker the files belong to `alumno`, and `root` owns none of them.
@@check
SORT_OUTPUT=1
ARGS=('alumno work' 'root work' 'alumno empty' 'alumno "work/sub"')
COMPARE="stdout exit"
@@solution
#!/bin/bash
find "$2" -type f -user "$1"

@@readme

#!/bin/bash
if (( $# != 2 )); then echo "Error: a user and a directory are needed" >&2; echo "Usage: $0 user dir" >&2; exit 1; fi
[[ -e $2 ]] || { echo "Error: $2 does not exist" >&2; exit 2; }
[[ -d $2 ]] || { echo "Error: $2 is not a directory" >&2; exit 3; }
id -u "$1" >/dev/null 2>&1 || { echo "Error: user $1 does not exist" >&2; exit 4; }
n=0
while IFS= read -r f; do echo "$f"; n=$((n + 1)); done < <(find "$2" -type f -user "$1")
echo "Files of $1: $n"
