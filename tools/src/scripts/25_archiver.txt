@@step 1 | Make the archive
@@readme
Write `archiver.sh DIR`. It creates `$HOME/archives/NAME.tgz`, a gzip-compressed tar archive of the directory `DIR`, where `NAME` is the name of `DIR` (without its path); `$HOME/archives` already exists in this step. Inside the archive the files are stored under `NAME/`. Print `Created <full path of the archive>`.

`tar czf archive.tgz -C "$(dirname dir)" "$(basename dir)"` stores the directory under its own name.
@@check
ARGS=('proj $(pre_arch)' '"my docs" $(pre_arch)' 'empty $(pre_arch)')
COMPARE="stdout exit files"
@@solution
#!/bin/bash
name=$(basename "$1")
out=$HOME/archives/$name.tgz
tar czf "$out" -C "$(dirname "$1")" "$name"
echo "Created $out"
@@step 2 | Create the archives directory
@@readme
If `$HOME/archives` does not exist, create it and print exactly `Directory <full path> created` before the `Created` line.
@@check
ARGS=('proj $(pre_arch)' 'proj' '"my docs"' 'empty')
COMPARE="stdout exit files"
@@solution
#!/bin/bash
dest=$HOME/archives
if [[ ! -d $dest ]]; then mkdir -p "$dest"; echo "Directory $dest created"; fi
name=$(basename "$1")
out=$dest/$name.tgz
tar czf "$out" -C "$(dirname "$1")" "$name"
echo "Created $out"

#!/bin/bash
if (( $# != 1 )); then echo "Error: one directory is needed" >&2; echo "Usage: $0 dir" >&2; exit 1; fi
[[ -e $1 ]] || { echo "Error: $1 does not exist" >&2; exit 2; }
[[ -d $1 ]] || { echo "Error: $1 is not a directory" >&2; exit 3; }
dest=$HOME/archives
if [[ ! -d $dest ]]; then mkdir -p "$dest"; echo "Directory $dest created"; fi
name=$(basename "$1")
out=$dest/$name.tgz
if [[ -e $out ]]; then echo "Error: $out already exists" >&2; exit 4; fi
tar czf "$out" -C "$(dirname "$1")" "$name"
echo "Created $out"
