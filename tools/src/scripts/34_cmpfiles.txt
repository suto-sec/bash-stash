@@fixture
setup() {
  printf 'one\ntwo\nthree\n' > a.txt; printf 'one\ntwo\nthree\n' > same.txt; printf 'one\nTWO\nthree\n' > diff.txt
  printf 'one\n' > short.txt; : > empty.txt; echo x > "my file.txt"; mkdir adir
}
usage_ok() { [[ $ERR == *sage* || $ERR == *Uso* || $ERR == *uso* || $ERR == *cmpfiles.sh* ]]; }
@@step 1 | Same or different
@@readme
Write `cmpfiles.sh FILE1 FILE2`. It prints `same` when the two files have exactly the same content and `different` otherwise. Exit code 0 in both cases.

`cmp -s a b` succeeds silently when the files are identical.
@@check
ARGS=('a.txt same.txt' 'a.txt diff.txt' 'a.txt short.txt' 'empty.txt empty.txt' '"my file.txt" a.txt')
COMPARE="stdout exit"
@@solution
#!/bin/bash
if cmp -s "$1" "$2"; then echo same; else echo different; fi
@@step 2 | Check the arguments
@@readme
Checks, in this order (errors on standard error, nothing on standard output): not exactly two arguments → error message **and the correct usage** (e.g. `Usage: cmpfiles.sh file1 file2`), exit **1**; the first argument is not a regular file → error with its name, exit **2**; the same for the second argument.
@@check
ARGS=('a.txt same.txt' 'a.txt diff.txt' '' 'a.txt' 'a.txt same.txt diff.txt' 'nothing.txt a.txt' 'a.txt nothing.txt' 'adir a.txt' 'a.txt adir')
COMPARE="stdout exit errmsg"
extra_check() {
  [[ $REF_CODE == 1 ]] && { usage_ok || fail "the message should show the correct usage"; }
  if [[ $REF_CODE == 2 ]]; then local a; a=$(eval "set -- $CASE"; [[ -f $W/$1 ]] && echo "$2" || echo "$1"); mentions "$a"; fi
}
@@solution
#!/bin/bash
if (( $# != 2 )); then echo "Error: two files are needed" >&2; echo "Usage: $0 file1 file2" >&2; exit 1; fi
for f in "$1" "$2"; do
  [[ -f $f ]] || { echo "Error: $f is not a regular file" >&2; exit 2; }
done
if cmp -s "$1" "$2"; then echo same; else echo different; fi

#!/bin/bash
if (( $# != 2 )); then echo "Error: two files are needed" >&2; echo "Usage: $0 file1 file2" >&2; exit 1; fi
for f in "$1" "$2"; do
  [[ -f $f ]] || { echo "Error: $f is not a regular file" >&2; exit 2; }
done
if cmp -s "$1" "$2"; then echo same; exit 0; fi
echo "different: $(wc -l < "$1") lines, $(wc -l < "$2") lines"
exit 4
