@@step 1 | Size of each subdirectory
@@readme
Write `diskuse.sh DIR`. For each **subdirectory directly inside** `DIR` (not files) print `NAME: SIZE`, where `SIZE` is the output of `du -sb` for it (bytes, including the contents). The order does not matter.

`du -sb path | cut -f1` gives just the number.
@@check
SORT_OUTPUT=1
ARGS=('base' 'empty')
COMPARE="stdout exit"
@@solution
#!/bin/bash
for d in "$1"/*/; do
  [[ -d $d ]] || continue
  d=${d%/}
  echo "$(basename "$d"): $(du -sb "$d" | cut -f1)"
done
exit 0
@@step 2 | Biggest first
@@readme
Print them from the **biggest to the smallest** size; for the same size, by name in alphabetical order (`bravo` and `delta` have the same size in the checker).
@@check
ARGS=('base' 'empty')
COMPARE="stdout exit"
@@solution
#!/bin/bash
for d in "$1"/*/; do
  [[ -d $d ]] || continue
  d=${d%/}
  echo "$(basename "$d"): $(du -sb "$d" | cut -f1)"
done | sort -t: -k2,2nr -k1,1
exit 0

  [[ -d $d ]] || continue
  d=${d%/}
  echo "$(basename "$d"): $(du -sb "$d" | cut -f1)"

@@step 4 | A total
@@readme
Finish with `Total: S`, the sum of the sizes printed (`Total: 0` when there are no subdirectories).
@@check
ARGS=('base' 'empty' '' 'nothing' 'notadir.txt')
COMPARE="stdout exit errmsg"
extra_check() {
  [[ $REF_CODE == 1 ]] && { usage_ok || fail "the message should show the correct usage"; }
  [[ $REF_CODE == [23] ]] && mentions "$(eval "set -- $CASE"; echo "$1")"

@@solution
#!/bin/bash
if (( $# != 1 )); then echo "Error: one directory is needed" >&2; echo "Usage: $0 dir" >&2; exit 1; fi
[[ -e $1 ]] || { echo "Error: $1 does not exist" >&2; exit 2; }
[[ -d $1 ]] || { echo "Error: $1 is not a directory" >&2; exit 3; }
total=0 lines=
for d in "$1"/*/; do
  [[ -d $d ]] || continue
  d=${d%/}
  s=$(du -sb "$d" | cut -f1)
  lines+="$(basename "$d"): $s"$'\n'
  total=$((total + s))
done
[[ -n $lines ]] && printf '%s' "$lines" | sort -t: -k2,2nr -k1,1
echo "Total: $total"
