@@script s14 | userinfo | Looking up a user | userinfo.sh
@@level 2
@@tags text, files, exit codes
@@cmds grep, cut -d: -f, test, exit codes
@@fixture
setup() {
  mkfl users.txt "root:x:0:0:root:/root:/bin/bash" "ana:x:1001:100:Ana Ruiz:/home/ana:/bin/zsh" "luis:x:1002:100:Luis,,,:/home/luis:/bin/bash" "anaelle:x:1003:100::/home/anaelle:/bin/sh" "daemon:x:1:1:daemon:/usr/sbin:/usr/sbin/nologin"
  : > emptyfile.txt; mkdir adir
}
usage_ok() { [[ $ERR == *sage* || $ERR == *Uso* || $ERR == *uso* || $ERR == *userinfo.sh* ]]; }
@@step 1 | The home directory
@@readme
Write `userinfo.sh FILE USER`. `FILE` has one user per line in the `/etc/passwd` layout (`name:x:uid:gid:comment:home:shell`). Print the **home directory** of `USER` (6th field). Only an exact name matches: `ana` is not `anaelle`.

`grep "^ana:" file` selects the line and `cut -d: -f6` the field.
@@check
ARGS=('users.txt ana' 'users.txt root' 'users.txt anaelle' 'users.txt luis')
COMPARE="stdout exit"
@@solution
#!/bin/bash
grep "^$2:" "$1" | cut -d: -f6
@@step 2 | More fields
@@readme
Print a single line `USER: uid=UID shell=SHELL` instead (the user name, the 3rd field and the 7th).
@@check
ARGS=('users.txt ana' 'users.txt root' 'users.txt anaelle' 'users.txt daemon')
COMPARE="stdout exit"
@@solution
#!/bin/bash
line=$(grep "^$2:" "$1")
uid=$(echo "$line" | cut -d: -f3)
shell=$(echo "$line" | cut -d: -f7)
echo "$2: uid=$uid shell=$shell"
@@step 3 | Check the arguments and the file
@@readme
Checks, in this order: not exactly two arguments → error message **and the correct usage**, exit **1**; `FILE` is not a readable regular file → error with its name, exit **2**.
@@check
ARGS=('users.txt ana' 'users.txt root' '' 'users.txt' 'users.txt ana x' 'nothing.txt ana' 'adir ana')
COMPARE="stdout exit errmsg"
extra_check() {
  [[ $REF_CODE == 1 ]] && { usage_ok || fail "the message should show the correct usage"; }
  [[ $REF_CODE == 2 ]] && mentions "$(eval "set -- $CASE"; echo "$1")"
}
@@solution
#!/bin/bash
if (( $# != 2 )); then echo "Error: a file and a user are needed" >&2; echo "Usage: $0 file user" >&2; exit 1; fi
[[ -f $1 && -r $1 ]] || { echo "Error: cannot read $1" >&2; exit 2; }
line=$(grep "^$2:" "$1")
uid=$(echo "$line" | cut -d: -f3)
shell=$(echo "$line" | cut -d: -f7)
echo "$2: uid=$uid shell=$shell"
@@step 4 | An unknown user
@@readme
If `USER` is not in the file, print an error message that includes the name on standard error and exit with code **3** (nothing on standard output).
@@check
ARGS=('users.txt ana' 'users.txt luis' 'users.txt nobody' 'users.txt an' 'emptyfile.txt ana' '' 'nothing.txt ana')
COMPARE="stdout exit errmsg"
extra_check() {
  [[ $REF_CODE == 1 ]] && { usage_ok || fail "the message should show the correct usage"; }
  [[ $REF_CODE == 2 ]] && mentions "$(eval "set -- $CASE"; echo "$1")"
  [[ $REF_CODE == 3 ]] && mentions "$(eval "set -- $CASE"; echo "$2")"
}
@@solution
#!/bin/bash
if (( $# != 2 )); then echo "Error: a file and a user are needed" >&2; echo "Usage: $0 file user" >&2; exit 1; fi
[[ -f $1 && -r $1 ]] || { echo "Error: cannot read $1" >&2; exit 2; }
line=$(grep "^$2:" "$1")
[[ -n $line ]] || { echo "Error: user $2 not found" >&2; exit 3; }
uid=$(echo "$line" | cut -d: -f3)
shell=$(echo "$line" | cut -d: -f7)
echo "$2: uid=$uid shell=$shell"
