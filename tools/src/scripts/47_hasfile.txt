  mkdir -p site/css site/img; echo a > site/index.html; echo b > site/about.html; echo c > "site/two words.txt"; echo d > site/css/main.css

@@step 1 | Yes or no
@@readme
Write `hasfile.sh DIR NAME...`. For every `NAME` print `NAME: yes` if `DIR/NAME` exists (a file or a directory) and `NAME: no` otherwise, in the order given.
@@check
ARGS=('site index.html' 'site index.html missing.txt css "two words.txt"' 'site nothing' 'empty a b')
COMPARE="stdout exit"
@@solution
#!/bin/bash
dir=$1; shift
for name in "$@"; do
  if [[ -e $dir/$name ]]; then echo "$name: yes"; else echo "$name: no"; fi
done

#!/bin/bash
if (( $# < 2 )); then echo "Error: a directory and a name are needed" >&2; echo "Usage: $0 dir name..." >&2; exit 1; fi

#!/bin/bash
if (( $# < 2 )); then echo "Error: a directory and a name are needed" >&2; echo "Usage: $0 dir name..." >&2; exit 1; fi
[[ -e $1 ]] || { echo "Error: $1 does not exist" >&2; exit 2; }
[[ -d $1 ]] || { echo "Error: $1 is not a directory" >&2; exit 3; }
dir=$1; shift
miss=0
for name in "$@"; do
  p=$dir/$name
  if [[ -f $p ]]; then echo "$name: file"
  elif [[ -d $p ]]; then echo "$name: directory"
  elif [[ -e $p ]]; then echo "$name: other"
  else echo "$name: no"; miss=$((miss + 1)); fi
done
if (( miss > 0 )); then echo "Missing: $miss"; exit 4; fi
