@@step 1 | One link per file
@@readme
Write `linkfarm.sh DIR LINKDIR`. In the existing directory `LINKDIR` it creates a **symbolic link** to every **regular file directly inside** `DIR`, with the same name; the link must point to the file's **absolute path** (`$(readlink -f "$f")`). Print `linked NAME` for each; the order does not matter. Assume no name exists yet in `LINKDIR`.
@@check

@@solution
#!/bin/bash
for f in "$1"/*; do
  [[ -f $f ]] || continue
  ln -s "$(readlink -f "$f")" "$2/$(basename "$f")"
  echo "linked $(basename "$f")"
done
exit 0

#!/bin/bash
if (( $# != 2 )); then echo "Error: a directory and a link directory are needed" >&2; echo "Usage: $0 dir linkdir" >&2; exit 1; fi
[[ -e $1 ]] || { echo "Error: $1 does not exist" >&2; exit 2; }
[[ -d $1 ]] || { echo "Error: $1 is not a directory" >&2; exit 3; }
if [[ -e $2 && ! -d $2 ]]; then echo "Error: $2 is not a directory" >&2; exit 3; fi
if [[ ! -d $2 ]]; then mkdir -p "$2"; echo "Directory $2 created"; fi
n=0 bad=0
for f in "$1"/*; do
  [[ -f $f ]] || continue
  name=$(basename "$f")
  if [[ -e $2/$name || -L $2/$name ]]; then echo "skipped $name" >&2; bad=1; continue; fi
  ln -s "$(readlink -f "$f")" "$2/$name"
  echo "linked $name"; n=$((n + 1))
done
echo "Linked $n files"
(( bad == 0 )) || exit 4
