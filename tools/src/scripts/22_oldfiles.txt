#!/bin/bash
if (( $# < 1 || $# > 2 )); then echo "Error: wrong number of arguments" >&2; echo "Usage: $0 days [dir]" >&2; exit 1; fi
[[ $1 =~ ^[0-9]+$ ]] || { echo "Error: '$1' is not a number of days" >&2; exit 2; }
dir=${2:-.}
[[ -d $dir ]] || { echo "Error: $dir is not a directory" >&2; exit 3; }
dest=$HOME/old
if [[ ! -d $dest ]]; then mkdir -p "$dest"; echo "Directory $dest created"; fi
while IFS= read -r -d '' f; do
  mv -f -- "$f" "$dest/"
done < <(find "$dir" -type f -mtime +"$1" -print0)

#!/bin/bash
if (( $# < 1 || $# > 2 )); then echo "Error: wrong number of arguments" >&2; echo "Usage: $0 days [dir]" >&2; exit 1; fi
[[ $1 =~ ^[0-9]+$ ]] || { echo "Error: '$1' is not a number of days" >&2; exit 2; }
dir=${2:-.}
[[ -d $dir ]] || { echo "Error: $dir is not a directory" >&2; exit 3; }
dest=$HOME/old
if [[ ! -d $dest ]]; then mkdir -p "$dest"; echo "Directory $dest created"; fi
n=0
while IFS= read -r -d '' f; do
  if mv -f -- "$f" "$dest/" 2>/dev/null; then n=$((n + 1)); else echo "could not move $f" >&2; fi
done < <(find "$dir" -type f -mtime +"$1" -print0)
echo "Moved $n files"
