@@ex 1340 at_intro | "$@" keeps arguments intact
@@level 1
@@cmds "$@", for
@@readme
Print every argument on its own line, prefixed with `- `. Use `"$@"` so arguments with spaces
stay whole.
@@check
SEEDS=1
ARGS=('' 'uno' 'uno dos tres' '"a b" c')
@@solution
#!/bin/bash
for a in "$@"; do
  echo "- $a"
done

@@ex 1341 shift_intro | shift
@@level 1
@@cmds shift
@@readme
Print `Primero: <$1>`. Then `shift` and print `Ahora: <new $1>`, or `none` if there isn't one.
@@check
SEEDS=1
ARGS=('' 'uno' 'uno dos' 'uno dos tres')
@@solution
#!/bin/bash
echo "Primero: $1"
shift
if [ $# -eq 0 ]; then echo none; else echo "Ahora: $1"; fi
