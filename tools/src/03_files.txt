@@ex 0305 rm_globs | Deleting with wildcards
@@level 2
@@cmds rm, *, ?, [...]
@@readme
The current directory contains many `Trimestre.*` files. Using **wildcards** (not listing names one by one):

1. Delete every file whose name is `Trimestre.17` followed by anything.
2. Delete the files named `Trimestre.18.` followed by a `1` or a `2` and ending in `.txt`.
3. Delete every file whose name has exactly **one character** before `.log` (e.g. `a.log`, not `ab.log`).

Leave everything else untouched. Use at most three `rm` commands (one per item).
@@check
COMPARE="stdout exit files"
setup() {
  local y q; for y in 16 17 18 19; do for q in 1 2 3 4; do touch "Trimestre.$y.$q.txt"; done; done
  touch Trimestre.17.todos.txt Trimestre.17.antiguos.txt Trimestre.18.1.bak Trimestre.18.12.txt Trimestre.170.txt
  touch "$(pick a b c).log" "$(pick x y z).log" "ab.log" "abc.log" ".log" "$(randr 1 9).log"
}
extra_check() { (( $(ans_code | wc -l) <= 3 )) || fail "use at most 3 rm commands (wildcards!)"; }
@@solution
#!/bin/bash
rm Trimestre.17*
rm Trimestre.18.[12].txt
rm ?.log

@@ex 0309 links_after_delete | What survives a deletion
@@level 3
@@cmds ln, ln -s, rm, cat, test
@@readme
The file `original` exists in the current directory.

1. Create a hard link `duro` and a symbolic link `blando` (target: `original`).
2. Delete `original`.
3. Print the content of `duro`.
4. Print `broken` if `blando` is a broken link (`[ -e blando ]` follows the link), `ok` otherwise.
5. Print the link count of `duro` (`stat -c %h`).

Think about **why** the hard link survives and the symbolic one does not.
@@check
COMPARE="stdout exit files"
setup() { randtext 3 > original; }
@@solution
#!/bin/bash
ln original duro
ln -s original blando
rm original
cat duro
if [ -e blando ]; then echo ok; else echo broken; fi
stat -c %h duro

@@ex 0310 stat | File metadata with stat
@@level 2
@@cmds stat -c
@@readme
For each of the files `a`, `b` and `c` in the current directory, print a line with:
name, size in bytes, permissions in octal and number of hard links, separated by spaces:

```
a 1532 644 1
b 12 600 2
c 0 755 1
```

Use `stat -c` with the right format sequences (`man stat`).
@@check
setup() { bigfile a $(randr 0 3000); bigfile b $(randr 0 50); bigfile c $(randr 0 9); chmod "$(pick 600 640 644 700)" a; chmod "$(pick 755 750 444)" b; [[ $(rand 2) == 1 ]] && ln c c2; true; }
extra_check() { must_use stat; }
@@solution
#!/bin/bash
stat -c '%n %s %a %h' a b c

@@ex 0311 basename_dirname | basename and dirname

@@ex 0314 readlink_f | Following link chains

@@check
setup() { mkdir -p real; randtext 1 > real/file; local n; n=$(randr 2 4); ln -s real/file "link$n"; local i; for ((i=n-1;i>=1;i--)); do ln -s "link$((i+1))" "link$i"; done; }

@@ex 0315 rm_r | Recursive deletion and safe rmdir
@@level 3
@@cmds rm -r, rmdir, 2>/dev/null
@@readme
1. Delete the directory `pruebatar` and **all** its content with a single command.
2. The directory `vacios` contains several subdirectories: some are empty and some are not.
   Remove **only the empty ones** with one `rmdir` command, hiding its error messages
   (`2>/dev/null`). The script must end with exit code **0**.
@@check
COMPARE="stdout exit files"
setup() { mkdir -p pruebatar/a/b; randtext 2 > pruebatar/a/b/f; local i d; for i in 1 2 3 4 5; do d="vacios/$(word)$i"; mkdir -p "$d"; (( i == 1 || $(rand 2) )) && touch "$d/x"; done; mkdir -p vacios/empty0; }
@@solution
#!/bin/bash
rm -r pruebatar
rmdir vacios/* 2>/dev/null
exit 0
