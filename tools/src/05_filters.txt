@@ex 0507 cut_chars | cut by characters and output delimiter
@@level 2
@@cmds cut -c, cut --output-delimiter
@@readme
`ls_output.txt` contains lines from `ls -l`. Print:

1. the first 10 characters of every line (the type and permissions)
2. a line `---`
3. the fields 1 and 3 of the file `notas.csv` (separated by `;`), but printed separated by ` | `
   (see `--output-delimiter`)
@@check
setup() {
  local i; for i in $(seq 6); do echo "$(pick - d l)$(pick rwx rw- r--)$(pick r-x r-- ---)$(pick r-x r-- ---) 1 $(word) $(word) $(randr 10 9999) Jun $(randr 10 28) 12:00 $(word)$i"; done > ls_output.txt

@@solution

cut -d';' -f1,3 --output-delimiter=' | ' notas.csv

@@ex 0513 tr_delete_squeeze | tr -d and tr -s

@@ex 0523 cut_sort_combo | Combining filters
@@level 3
@@cmds cut, sort, head, tr
@@readme
`ventas.csv` has a header line and then lines `fecha,producto,unidades,precio`.
Print the **3 products with most units** in one sale (lines, not aggregated), as `producto unidades`
separated by a single space, highest first (ties: product name alphabetically).
The header must be ignored.
@@check
setup() { echo "fecha,producto,unidades,precio" > ventas.csv; local i; for i in $(seq 12); do echo "2026-0$(randr 1 9)-1$(randr 0 9),$(word),$(randr 1 300),$(randr 1 99).$(randr 10 99)"; done >> ventas.csv; }

@@ex 0524 word_freq | Word frequency
@@level 4
@@cmds tr, sort, uniq -c, head
@@readme
Print the **5 most frequent words** of `articulo.txt` with their count, in `uniq -c` format,
most frequent first. For equal counts, alphabetical order of the word.

- words are sequences of letters; convert everything to lowercase first
- anything that is not a letter separates words

Hint: `tr -cs 'A-Za-z' '\n'` puts every word on its own line.
@@check
setup() { local i; for i in $(seq 25); do echo "$(word) $(word), $(pick The the THE Linux kernel) $(word). $(pick shell Shell SHELL)!"; done > articulo.txt; }
@@solution
#!/bin/bash
tr 'A-Z' 'a-z' < articulo.txt | tr -cs 'a-z' '\n' | grep -v '^$' | sort | uniq -c | sort -k1,1nr -k2,2 | head -n 5
