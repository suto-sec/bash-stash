@@topic 07_find | find
@@ex 0701 find_name | Searching by name
@@level 1
@@cmds find -name
@@readme
Under the directory `lib` (recursively), find the files whose name starts with `ls` and ends in `.so`.
Print their paths **sorted** (find does not guarantee any order: pipe it to `sort`).

Remember to **quote** the pattern so the shell does not expand it.
@@check
setup() { mkdir -p lib/{a,b/c}; local i; for i in 1 2 3 4; do touch "lib/$(pick a b b/c)/ls$(word)$i.so" "lib/$(pick a b)/$(word)ls$i.so" "lib/ls$(word)$i.so.1"; done; touch lsfake.so; }
@@solution
#!/bin/bash
find lib -name 'ls*.so' | sort

@@ex 0704 find_size | Searching by size

echo ---

echo ---

@@ex 0706 find_perm | Searching by permissions
@@level 3
@@cmds find -perm -mode, -perm /mode
@@readme
Under `bin`, print sorted, separated by `---`:

1. regular files with **exactly** permissions `755`
2. regular files where **someone** (user, group or others) has **execute** permission (`-perm /111`)
3. regular files that have **all** of `u+w` and `g+w` (`-perm -220`)
4. regular files **without any** execute permission
@@check
setup() { mkdir -p bin/sub; local i f; for i in $(seq 10); do f="bin/$(pick . sub)/$(word)$i$(pick .sh .bin '')"; touch "$f"; chmod "$(pick 755 644 600 700 664 775 711 666 744 601)" "$f"; done; }
@@solution
#!/bin/bash
find bin -type f -perm 755 | sort
echo ---
find bin -type f -perm /111 | sort
echo ---
find bin -type f -perm -220 | sort
echo ---
find bin -type f ! -perm /111 | sort

@@ex 0709 find_logic | Combining conditions: ! -o ( )
@@level 3
@@cmds find !, -o, \( \)
@@readme
Under `docs`, print sorted all files **and** directories whose name starts
with `a` **or** `b` and does **not** contain the character `~`.

Careful: `-o` has lower precedence than the implicit AND; you need `\( ... -o ... \)`.
@@check
setup() { mkdir -p docs/{alfa,beta~,c}; local i; for i in $(seq 10); do touch "docs/$(pick . alfa c)/$(pick a b c A)$(word)$i$(pick '' '~' '.txt')"; done; }
@@solution
#!/bin/bash
find docs \( -name 'a*' -o -name 'b*' \) ! -name '*~*' | sort

@@ex 0715 find_exam | The exam filter
@@level 4
@@cmds find -type f -perm /111 \( -name -o -name \)
@@readme
This is the core of the exam-style deploy script (1801). Under `src` (recursively), print sorted the **regular files**
(not directories) that have **some** execute permission enabled (user, group or other) and whose
name ends in `.sh` **or** `.bin`.
@@check
setup() {
  mkdir -p "src/a/b" "src/c d"; local i f
  for i in $(seq 12); do f="src/$(pick . a a/b 'c d')/$(word)$i$(pick .sh .bin .shx .txt .sh.bak)"; touch "$f"; chmod "$(pick 755 644 700 604 601 610 744)" "$f"; done
  mkdir -p src/dir.sh; chmod 755 src/dir.sh
}
@@solution
#!/bin/bash
find src -type f -perm /111 \( -name '*.sh' -o -name '*.bin' \) | sort

@@ex 0716 find_count_ext | Counting by extension
@@level 4
@@cmds find, sed/rev, sort, uniq -c
@@readme
Under `repo`, count the regular files **by extension** (the part after the last `.` of the name;
files without a dot are ignored). Print as `uniq -c` does, sorted by count descending and then by
extension alphabetically.

Hint: `find repo -type f -name '*.*'` + extract the extension (e.g. `sed 's/.*\.//'`).
@@check
setup() { mkdir -p repo/{a,b/c}; local i; for i in $(seq 20); do touch "repo/$(pick . a b/c)/$(word)$i$(pick .c .h .c .md .txt .tar.gz '' .c)"; done; }
@@solution
#!/bin/bash
find repo -type f -name '*.*' | sed 's/.*\.//' | sort | uniq -c | sort -k1,1nr -k2,2
